cmake_minimum_required(VERSION 3.20)
project(pct_ba LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Ceres REQUIRED)   # static build made by build.sh (no SuiteSparse, no glog)
add_library(pct_ba SHARED pct_ba.cc)
target_link_libraries(pct_ba PRIVATE Ceres::ceres)
if(APPLE)
  set_target_properties(pct_ba PROPERTIES INSTALL_NAME_DIR "@rpath" MACOSX_RPATH ON)
  target_link_options(pct_ba PRIVATE -Wl,-dead_strip)
elseif(UNIX)
  target_link_options(pct_ba PRIVATE -Wl,--exclude-libs,ALL -Wl,--gc-sections)
endif()
install(TARGETS pct_ba LIBRARY DESTINATION . RUNTIME DESTINATION .)
